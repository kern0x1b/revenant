// Configuring the accessibility features of a Single App Mode session, iOS 12.2, and the
// error domain its failures are reported in (the value of the domain read from the 13.4
// cache, the oldest held release that exports it).
//
// It is in this object because it reads UIGuidedAccessErrorDomain, which is in this object too: a C
// function that reaches a constant in another object breaks in exactly the bands where the two land
// on opposite sides of the release split (toolchain/AGENTS.md, "A C function shared between backport
// files"). Both are first exported by the held 13.4 cache, so the object carries one release.

void UIGuidedAccessConfigureAccessibilityFeatures(UIGuidedAccessAccessibilityFeature features, BOOL enabled, void (^completion)(BOOL success, NSError *error))
{
    if (!completion)
        return;
    // The header says a caller without rights for an action gets this error, for example when it has
    // asked for a configuration change without being locked into Single App Mode by a managed profile (SDK 26.2,
    // UIGuidedAccess.h:17-19, UIGuidedAccessErrorPermissionDenied) - which is
    // this case exactly: the function changes what a Single App Mode session allows, there is no
    // such session here, and the request is answered on the next turn of the main queue, as the
    // release's own completion is not called from inside the request.
    dispatch_async(dispatch_get_main_queue(), ^{
        completion(NO, [NSError errorWithDomain:UIGuidedAccessErrorDomain
                                            code:UIGuidedAccessErrorPermissionDenied
                                        userInfo:nil]);
    });
}
